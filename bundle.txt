function updateNavigation(route) {
  document.querySelectorAll('.sidebar__link').forEach((link) => {
    const routePath = route?.path
    const linkPath = link.hash.replace(/^#/, '')
    const active = routePath === '/subjects' && linkPath === '/subjects'
      ? true
      : routePath === '/subjects/:id' && linkPath === '/subjects'
        ? true
        : linkPath === '/gwa' && (routePath === '/gwa' || routePath === '/gwa-calculator')
          ? true
        : linkPath === routePath
    if (active) link.setAttribute('aria-current', 'page')
    else link.removeAttribute('aria-current')
  })
  closeMobileDrawer(false)
}

import { getSession } from './auth/auth.js'
import { session } from './state/session.js'
import { isUuid } from './utils/validate.js'

function matchRoute(hashPath, route) {
  const actual = hashPath.split('/').filter(Boolean)
  const expected = route.path.split('/').filter(Boolean)
  if (actual.length !== expected.length) return null

  const params = {}
  for (let index = 0; index < expected.length; index += 1) {
    const segment = expected[index]
    if (segment.startsWith(':')) {
      params[segment.slice(1)] = decodeURIComponent(actual[index])
    } else if (segment !== actual[index]) {
      return null
    }
  }
  return params
}

export function startRouter({ routeTable, pageMap, container, notFound, onNavigate, onAuthRequired }) {
  let activeCleanup = null
  let navigationId = 0

  async function navigate() {
    const currentNavigation = ++navigationId
    const hash = window.location.hash || '#/login'
    const hashPath = `/${hash.replace(/^#\/?/, '').replace(/\/+$/, '')}`
    let match = null

    for (const route of routeTable) {
      const params = matchRoute(hashPath, route)
      if (params) {
        match = { route, params }
        break
      }
    }

    if (activeCleanup) {
      activeCleanup()
      activeCleanup = null
    }

    if (!match) {
      await notFound(container, { path: hashPath })
      onNavigate?.({ route: null, params: {}, path: hashPath })
      return
    }

    if (match.route.requiresAuth) {
      let user = null
      try {
        user = await getSession()
      } catch (error) {
        onAuthRequired?.(error)
      }
      if (currentNavigation !== navigationId) return
      if (!user) {
        window.location.hash = '#/login'
        return
      }
    }
    if (!match.route.requiresAuth && ['login', 'register', 'forgot-password'].includes(match.route.name)) {
      let user = null
      try {
        user = await getSession()
      } catch (error) {
        onAuthRequired?.(error)
      }
      if (currentNavigation !== navigationId) return
      if (user) {
        window.location.hash = session.recoveryPending ? '#/reset-password' : '#/dashboard'
        return
      }
    }

    for (const [key, rule] of Object.entries(match.route.paramRules || {})) {
      if (rule === 'uuid' && !isUuid(match.params[key])) {
        await notFound(container, { path: hashPath, invalidParam: key })
        onNavigate?.({ route: match.route, params: match.params, path: hashPath })
        return
      }
    }

    const render = pageMap[match.route.name]
    if (!render) {
      await notFound(container, { path: hashPath })
    } else {
      const cleanup = await render(container, match.params)
      if (typeof cleanup === 'function') activeCleanup = cleanup
    }
    onNavigate?.({ route: match.route, params: match.params, path: hashPath })
  }

  window.addEventListener('hashchange', navigate)
  navigate()

  return () => {
    window.removeEventListener('hashchange', navigate)
    if (activeCleanup) activeCleanup()
  }
}

function setDrawerOpen(open, trigger = null, restoreFocus = true) {
  const nav = document.querySelector('.sidebar')
  if (!nav) return
  const backdrop = document.querySelector('.sidebar-backdrop')
  nav.classList.toggle('sidebar--open', open)
  backdrop?.classList.toggle('sidebar-backdrop--visible', open)
  const hidden = !open && window.innerWidth <= 1023
  nav.inert = hidden
  if (hidden) nav.setAttribute('aria-hidden', 'true')
  else nav.removeAttribute('aria-hidden')
  const triggers = document.querySelectorAll('.top-bar__menu, .mobile-header .icon-button')
  triggers.forEach(button => button.setAttribute('aria-expanded', String(open)))
  if (open) nav.querySelector('a[href], button:not(:disabled), select:not(:disabled)')?.focus()
  else if (restoreFocus) (trigger || triggers[0])?.focus()
}

function installDrawerKeyboardHandler() {
  if (installDrawerKeyboardHandler.installed) return
  installDrawerKeyboardHandler.installed = true
  document.addEventListener('keydown', (event) => {
    const nav = document.querySelector('.sidebar')
    if (!nav?.classList.contains('sidebar--open') || window.innerWidth > 1023) return
    if (event.key === 'Escape') {
      event.preventDefault()
      setDrawerOpen(false)
      return
    }
    if (event.key !== 'Tab') return
    const focusable = Array.from(nav.querySelectorAll('a[href], button:not(:disabled), select:not(:disabled)'))
    if (!focusable.length) return
    const first = focusable[0]
    const last = focusable[focusable.length - 1]
    if (event.shiftKey && (document.activeElement === first || !nav.contains(document.activeElement))) {
      event.preventDefault()
      last.focus()
    } else if (!event.shiftKey && (document.activeElement === last || !nav.contains(document.activeElement))) {
      event.preventDefault()
      first.focus()
    }
  })
}

export function closeMobileDrawer(restoreFocus = true) {
  setDrawerOpen(false, null, restoreFocus)
}

export function sidebar() {
  const nav = h('aside', { className: 'sidebar' })
  nav.id = 'primary-sidebar'
  nav.setAttribute('aria-label', 'Sidebar')
  const updateDrawerVisibility = () => {
    const hidden = window.innerWidth <= 1023 && !nav.classList.contains('sidebar--open')
    nav.inert = hidden
    if (hidden) nav.setAttribute('aria-hidden', 'true')
    else nav.removeAttribute('aria-hidden')
  }
  updateDrawerVisibility()
  window.addEventListener('resize', updateDrawerVisibility)
  const brand = h('a', { className: 'sidebar__brand', href: '#/dashboard', text: 'GradeTrack' })
  const links = [
    ['Dashboard', '#/dashboard'], ['Semesters', '#/semesters'], ['Subjects', '#/subjects'],
    ['Assessments', '#/assessments'], ['Grade Calculator', '#/grade-calculator'],
    ['Predictor', '#/grade-predictor'], ['GWA', '#/gwa'],
    ['History', '#/history'], ['Analytics', '#/analytics'], ['Demo', '#/demo'], ['Settings', '#/settings']
  ]
  const list = h('nav', { className: 'sidebar__nav', ariaLabel: 'Primary navigation' })
  for (const [label, href] of links) {
    const link = h('a', { className: 'sidebar__link', href, text: label })
    link.addEventListener('click', () => {
      if (nav.classList.contains('sidebar--open')) setDrawerOpen(false)
    })
    list.appendChild(link)
  }
  const signOutButton = h('button', { className: 'button button--ghost', type: 'button', text: 'Sign out' })
  let signingOut = false
  signOutButton.addEventListener('click', async () => {
    if (signingOut) return
    signingOut = true
    signOutButton.disabled = true
    signOutButton.textContent = 'Signing out…'
    try {
      await signOut()
    } catch (error) {
      toast(authErrorMessage(error, 'logout'), 'error')
    } finally {
      signingOut = false
      signOutButton.disabled = false
      signOutButton.textContent = 'Sign out'
    }
  })
  nav.append(brand, list, signOutButton)
  return nav
}

export function topBar(title = 'GradeTrack') {
  installDrawerKeyboardHandler()
  const header = h('header', { className: 'top-bar' })
  const select = h('select', { 'data-semester-switcher': '', ariaLabel: 'Selected semester', disabled: true })
  header.append(
    h('div', { className: 'top-bar__title' }, [h('strong', { text: title })]),
    h('label', { className: 'semester-switcher' }, [
      h('span', { text: 'Semester' }),
      select
    ])
  )
  const menu = h('button', { className: 'icon-button top-bar__menu', type: 'button', ariaLabel: 'Open menu', ariaControls: 'primary-sidebar', ariaExpanded: 'false', text: '☰' })
  header.prepend(menu)
  menu.addEventListener('click', (event) => {
    const sidebarElement = document.querySelector('.sidebar')
    setDrawerOpen(!sidebarElement?.classList.contains('sidebar--open'), event.currentTarget)
  })
  let semesters = []
  select.addEventListener('change', async () => {
    const selected = semesters.find(item => item.id === select.value)
    if (!selected) return
    select.disabled = true
    try {
      session.profile = await updateProfile({ active_semester_id: selected.id })
      session.activeSemester = selected
      window.dispatchEvent(new CustomEvent('semester:changed', { detail: { semester: selected } }))
    } catch {
      toast('Could not save the selected semester.', 'error')
      select.value = session.activeSemester?.id || ''
    } finally {
      select.disabled = false
    }
  })

  header.refreshSemesterSwitcher = async () => {
    const previousId = session.activeSemester?.id
    try {
      const context = await getSemesterContext()
      semesters = context.semesters
      select.replaceChildren()
      if (!semesters.length) {
        select.appendChild(h('option', { value: '', text: 'No semesters yet' }))
        select.disabled = true
        return
      }
      for (const semester of semesters) {
        select.appendChild(h('option', {
          value: semester.id,
          text: `${semester.academic_year} · ${semester.term}${semester.status === 'active' ? ' · Active' : ' · Archived'}`
        }))
      }
      select.value = context.selected.id
      select.disabled = false
      if (previousId !== context.selected.id) {
        window.dispatchEvent(new CustomEvent('semester:changed', { detail: { semester: context.selected } }))
      }
    } catch {
      select.disabled = true
      toast('Could not load semesters.', 'error')
    }
  }
  window.addEventListener('semesters:changed', header.refreshSemesterSwitcher)
  return header
}

function token(name, fallback) {
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback
}

function collectActivity(subjects, semesterLabels) {
  const events = []
  for (const subject of subjects) {
    const semesterLabel = semesterLabels.get(subject.semester_id)
    const subjectLabel = `${subject.code} · ${subject.name}${semesterLabel ? ` · ${semesterLabel}` : ''}`
    if (subject.created_at) {
      events.push({
        date: subject.created_at,
        label: 'Subject added',
        detail: subjectLabel
      })
    }
    if (subject.final_grade?.updated_at) {
      events.push({
        date: subject.final_grade.updated_at,
        label: 'Final grade updated',
        detail: `${subjectLabel} · ${subject.final_grade.status || subject.final_grade.grade_value}`
      })
    }
    for (const assessment of subject.assessments ?? []) {
      const date = assessment.updated_at || assessment.created_at
      if (!date) continue
      events.push({
        date,
        label: assessment.is_completed ? 'Assessment completed' : 'Assessment updated',
        detail: `${subjectLabel} · ${assessment.name}`
      })
    }
  }
  return events
    .sort((left, right) => new Date(right.date).getTime() - new Date(left.date).getTime())
    .slice(0, 6)
}

function assessmentList(bucket) {
  const items = [
    ...bucket.overdue.map(assessment => ({ assessment, overdue: true })),
    ...bucket.upcoming.map(assessment => ({ assessment, overdue: false }))
  ]
  if (!items.length) {
    return h('div', { className: 'empty-state empty-state--compact' }, [
      h('strong', { text: 'No overdue or upcoming assessments' }),
      h('p', { text: 'Assessments due in the next 14 days will appear here.' })
    ])
  }
  const list = h('ul', { className: 'dashboard-assessment-list' })
  for (const { assessment, overdue } of items) {
    list.appendChild(h('li', { className: overdue ? 'dashboard-assessment dashboard-assessment--overdue' : 'dashboard-assessment' }, [
      h('div', {}, [
        h('strong', { text: assessment.name }),
        h('span', { className: 'text-muted', text: `${assessment.subjectCode} · ${assessment.subjectName}` })
      ]),
      h('span', { className: `badge${overdue ? ' badge--danger' : ''}`, text: `${overdue ? 'Overdue' : 'Upcoming'} · ${formatDate(assessment.due_date)}` })
    ]))
  }
  return list
}

}

export async function render(container) {
  const heading = h('div', { className: 'page-heading' }, [
    h('div', {}, [h('p', { className: 'eyebrow', text: 'Overview' }), h('h1', { text: 'Dashboard' })])
  ])
  container.replaceChildren(heading, h('div', { className: 'skeleton', ariaLabel: 'Loading dashboard' }))

  let records
  try {
    records = await loadAcademicRecords()
  } catch (error) {
    container.replaceChildren(heading, h('div', { className: 'empty-state' }, [
      h('strong', { text: 'Could not load your dashboard' }),
      h('p', { text: dataErrorMessage(error, 'Try again in a moment.') })
    ]))
    return
  }

  if (!records.semesters.length) {
    container.replaceChildren(
      heading,
      h('div', { className: 'empty-state dashboard-first-semester' }, [
        h('strong', { text: 'Create your first semester' }),
        h('p', { text: 'Add a semester to organize subjects, assessments, and grades.' }),
        h('a', { className: 'button button--primary', href: '#/semesters', text: 'Create your first semester' })
      ]),
      quickActions()
    )
    return
  }

  const activeSemester = records.semesters.find(item => item.id === records.profile?.active_semester_id)
    ?? records.semesters.find(item => item.status === 'active')
    ?? records.semesters[0]
  const subjects = records.semesterRecords.find(item => item.semester.id === activeSemester.id)?.subjects ?? []
  const semesterLabels = new Map(records.semesters.map(semester => [
    semester.id,
    `${semester.academic_year} · ${semester.term}`
  ]))
  const decimals = Number.isInteger(Number(records.profile?.gwa_decimals))
    ? Number(records.profile.gwa_decimals)
    : 2
  const gwa = computeGwa(subjects)
  const assessments = subjects.flatMap(subject =>
    (subject.assessments ?? []).map(assessment => ({
      ...assessment,
      subjectCode: subject.code,
      subjectName: subject.name
    }))
  )
  const bucket = classifyAssessments(assessments)
  const currentAverages = subjects.map(subject => ({
    subject,
    average: computeSubjectGrade(
      subject,
      subject.assessment_categories ?? [],
      subject.assessments ?? []
    ).currentAverage
  }))
  const completedAverages = currentAverages.filter(item => Number.isFinite(item.average))
  const stats = h('div', { className: 'stats-grid dashboard-stats' }, [
    metricCard('Semester GWA', formatGwa(gwa.gwa, decimals), `${activeSemester.academic_year} · ${activeSemester.term}`),
    metricCard('Total units', formatUnits(sumSubjectUnits(subjects))),
    metricCard('Subjects', String(subjects.length)),
    metricCard('Completed assessments', `${bucket.completed.length} / ${bucket.total}`),
    metricCard('Upcoming · 14 days', String(bucket.upcoming.length), `${bucket.overdue.length} overdue`)
  ])

  const overview = h('section', { className: 'card dashboard-overview' }, [
    h('div', { className: 'section-heading' }, [
      h('div', {}, [
        h('h2', { text: 'Subject overview' }),
        h('p', { className: 'text-muted', text: `${activeSemester.academic_year} · ${activeSemester.term}` })
      ])
    ])
  ])
  if (!subjects.length) {
    overview.appendChild(h('div', { className: 'empty-state empty-state--compact' }, [
      h('strong', { text: 'No subjects in this semester yet' }),
      h('p', { text: 'Add your first subject to start tracking performance.' }),
      h('a', { className: 'button button--primary', href: '#/subjects', text: 'Add a subject' })
    ]))
  } else {
    const list = h('div', { className: 'dashboard-subject-list' })
    for (const { subject, average } of currentAverages) {
      const currentAverage = Number.isFinite(average) ? formatPercent(average) : 'No completed assessments'
      const finalGrade = formatFinalGrade(subject.final_grade, 'No final grade')
      list.appendChild(h('a', { className: 'dashboard-subject', href: `#/subjects/${subject.id}` }, [
        h('span', { className: 'dashboard-subject__name' }, [
          h('strong', { text: subject.code }),
          h('span', { text: subject.name })
        ]),
        h('span', {}, [h('span', { className: 'dashboard-subject__label', text: 'Current average' }), h('strong', { text: currentAverage })]),
        h('span', {}, [h('span', { className: 'dashboard-subject__label', text: 'Final grade' }), h('strong', { text: finalGrade })]),
        h('span', { className: 'text-muted', text: formatUnits(Number(subject.units)) })
      ]))
    }
    overview.appendChild(list)
  }

  const performance = h('section', { className: 'card dashboard-chart-card' }, [
    h('h2', { text: 'Performance by subject' })
  ])
  let chart = null
  let canvas = null
  if (completedAverages.length) {
    const summary = completedAverages
      .map(({ subject, average }) => `${subject.code}: ${formatPercent(average)}`)
      .join(' · ')
    canvas = h('canvas', { role: 'img', ariaLabel: `Current subject averages. ${summary}` })
    performance.append(
      h('div', { className: 'chart-container' }, [canvas]),
      h('p', { className: 'chart-summary', text: `Current averages: ${summary}` })
    )
  } else {
    performance.appendChild(h('p', { className: 'text-muted', text: 'Complete an assessment to see subject performance here.' }))
  }

  const assessmentSection = h('section', { className: 'card dashboard-assessments' }, [
    h('div', { className: 'section-heading' }, [

  ])

    h('h2', { text: 'Recent activity' }),
    activityList(collectActivity(records.subjects, semesterLabels))
  ])
  const lowerGrid = h('div', { className: 'dashboard-lower-grid' }, [assessmentSection, activitySection])
  container.replaceChildren(heading, stats, overview, performance, lowerGrid, quickActions())
  if (canvas) {
    try {
      const { default: Chart } = await import('chart.js/auto')
      chart = new Chart(canvas, {
        type: 'bar',
        data: {
          labels: completedAverages.map(({ subject }) => subject.code),
          datasets: [{
            label: 'Current average (%)',
            data: completedAverages.map(({ average }) => average),
            backgroundColor: token('--accent', '#55c99a'),
            borderColor: token('--accent-strong', '#39ad7b'),
            borderWidth: 1
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: { legend: { display: false } },
          scales: {
            x: { ticks: { color: token('--text-muted', '#a8a8ad') }, grid: { color: token('--border', '#39393e') } },
            y: { beginAtZero: true, max: 100, ticks: { color: token('--text-muted', '#a8a8ad') }, grid: { color: token('--border', '#39393e') } }
          }
        }
      })
    } catch {
      performance.appendChild(h('p', { className: 'field-error', text: 'The performance chart could not load. The text summary above is still available.' }))
    }
  }
  return () => chart?.destroy()
}

===== src/js/utils/dom.js =====

export function h(tag, attributes = {}, children = []) {
  const element = document.createElement(tag)
  for (const [key, value] of Object.entries(attributes)) {
    if (value == null) continue
    if (key === 'className') element.className = value
    else if (key === 'text') element.textContent = value
    else if (key === 'html') throw new TypeError('Raw HTML is not supported by h().')
    else if (key.startsWith('aria') || key.startsWith('data')) {
      const prefix = key.startsWith('aria') ? 'aria' : 'data'
      const suffix = key.slice(prefix.length)
        .replace(/^-+/, '')
        .replace(/([A-Z])/g, '-$1')
        .replace(/^-+/, '')
        .toLowerCase()
      element.setAttribute(`${prefix}-${suffix}`, String(value))
    }
    else if (key in element) element[key] = value
    else element.setAttribute(key, String(value))
  }
  for (const child of Array.isArray(children) ? children : [children]) {
    if (child == null) continue
    element.appendChild(typeof child === 'string' ? document.createTextNode(child) : child)
  }
  return element
}

export function escapeHtml(value) {
  return String(value ?? '').replace(/[&<>"']/g, char => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;'
  })[char])
}


===== src/js/api/semestersApi.js =====

import { supabase } from '../supabase.js'
import { session } from '../state/session.js'
import { getProfile } from './profileApi.js'

function requireSupabase() {
  if (!supabase) throw new Error('Supabase is not configured.')
}

export async function listSemesters() {
  requireSupabase()
  const { data, error } = await supabase
    .from('semesters')
    .select('*')
    .order('start_date', { ascending: false })
    .order('academic_year', { ascending: false })
  if (error) throw error
  return data ?? []
}

export async function getSemesterContext() {
  const [semesters, profile] = await Promise.all([listSemesters(), getProfile()])
  const selected = semesters.find(item => item.id === profile?.active_semester_id)
    ?? semesters.find(item => item.status === 'active')
    ?? semesters[0]
    ?? null
  session.activeSemester = selected
  return { semesters, selected }
}

export async function createSemester(values) {
  requireSupabase()
  const { data, error } = await supabase.rpc('create_semester', {
    p_academic_year: values.academic_year,
    p_term: values.term,
    p_start_date: values.start_date,
    p_end_date: values.end_date || null,
    p_status: values.status || 'archived'
  })
  if (error) throw error
  return Array.isArray(data) ? data[0] : data
}

export async function updateSemester(id, values) {
  requireSupabase()
  const { data, error } = await supabase
    .from('semesters')
    .update(values)
    .eq('id', id)
    .select()
    .single()
  if (error) throw error
  return data
}

export async function setActiveSemester(id) {
  requireSupabase()
  const { data, error } = await supabase.rpc('set_active_semester', { p_semester_id: id })
  if (error) throw error
  const semester = Array.isArray(data) ? data[0] : data
  if (!semester) throw new Error('Could not set the active semester.')
  session.activeSemester = semester
  return semester
}

export async function archiveSemester(id) {
  const semester = await updateSemester(id, { status: 'archived' })
  if (session.activeSemester?.id === id) session.activeSemester = semester
  return semester
}

export async function deleteSemester(id) {
  requireSupabase()
  const { error } = await supabase.from('semesters').delete().eq('id', id)
  if (error) throw error
  if (session.activeSemester?.id === id) session.activeSemester = null
}

import { supabase } from '../supabase.js'
export async function getProfile() {
  if (!supabase) return null
  const { data, error } = await supabase.from('profiles').select('*').single()
  if (error && error.code !== 'PGRST116') throw error
  return data
}
export async function updateProfile(values) {
  const { data, error } = await supabase.from('profiles').update(values).eq('id', (await supabase.auth.getUser()).data.user.id).select().single()
  if (error) throw error
  return data
}
